cmake_minimum_required(VERSION 3.22)
project(middleware_anticheat LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
  message(FATAL_ERROR "Compila fuera de los fuentes: cmake -S . -B build -G Ninja")
endif()
if(NOT CMAKE_SYSTEM_NAME MATCHES "^(Linux|Windows)$")
  message(FATAL_ERROR "Plataformas implementadas: Linux y Windows")
endif()

include(CTest)


add_library(lab_options INTERFACE)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(lab_options INTERFACE
    -Wall -Wextra -Wshadow -Wstrict-prototypes -Wmissing-prototypes
    -Wcast-align -Wwrite-strings -fstack-protector-strong)
endif()

add_library(lab_contract STATIC lab/contracts/lab_util.c lab/contracts/evidence.c)
target_include_directories(lab_contract PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(lab_contract PUBLIC lab_options)

add_library(lab_game STATIC lab/game/game.c lab/game/view.c)
target_include_directories(lab_game PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(lab_game PUBLIC lab_options)
add_executable(lab_game_demo lab/app/game_demo.c)
target_link_libraries(lab_game_demo PRIVATE lab_game lab_contract)
add_executable(lab_game_host lab/app/game_host.c)
target_link_libraries(lab_game_host PRIVATE lab_game lab_contract)

add_executable(lab_app lab/app/labapp.c lab/app/worker.c)
if(WIN32)
  target_link_libraries(lab_contract PUBLIC bcrypt)
  add_library(lab_capabilities STATIC lab/platform/capabilities_windows.c)
  target_link_libraries(lab_capabilities PUBLIC lab_contract)
else()
  add_library(lab_host_status STATIC lab/platform/host_status.c)
  target_link_libraries(lab_host_status PUBLIC lab_options)
  add_executable(lab_host lab/app/hostcheck.c)
  target_link_libraries(lab_host PRIVATE lab_host_status)
  find_package(Threads REQUIRED)
  find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)
  add_executable(lab_ima_replay lab/app/ima_replay.c)
  target_link_libraries(lab_ima_replay PRIVATE lab_options OpenSSL::Crypto)
  add_library(lab_capabilities STATIC lab/platform/capabilities_linux.c)
  target_link_libraries(lab_capabilities PUBLIC lab_contract Threads::Threads OpenSSL::Crypto)
  add_library(lab_net STATIC lab/platform/net.c)
  target_link_libraries(lab_net PUBLIC lab_contract)
  foreach(component IN ITEMS server client adapter)
    add_executable(lab_${component} lab/${component}/${component}.c)
    target_link_libraries(lab_${component} PRIVATE lab_net)
  endforeach()
  target_link_libraries(lab_client PRIVATE lab_capabilities)
endif()
target_link_libraries(lab_app PRIVATE lab_capabilities)

option(LAB_TPM "Build the TPM quote decoder using the installed tpm2-tss MU library" OFF)
if(LAB_TPM)
  if(WIN32)
    message(FATAL_ERROR "The initial TPM experiment requires Linux")
  endif()
  find_path(TSS2_INCLUDE_DIR tss2/tss2_mu.h
    HINTS "${CMAKE_SOURCE_DIR}/results/toolchain/root/usr/include" REQUIRED)
  find_library(TSS2_MU_LIBRARY NAMES tss2-mu libtss2-mu.so.0 REQUIRED)
  add_executable(lab_quote_inspect lab/app/quote_inspect.c)
  target_include_directories(lab_quote_inspect PRIVATE "${TSS2_INCLUDE_DIR}")
  target_link_libraries(lab_quote_inspect PRIVATE lab_options "${TSS2_MU_LIBRARY}")
  if(BUILD_TESTING)
    find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
    find_program(TPM2_CHECKQUOTE_EXECUTABLE tpm2_checkquote REQUIRED)
    find_program(OPENSSL_EXECUTABLE openssl REQUIRED)
    add_test(NAME attestation_offline COMMAND "${Python3_EXECUTABLE}"
      "${CMAKE_SOURCE_DIR}/scripts/verificar-atestacion.py"
      --inspector $<TARGET_FILE:lab_quote_inspect>)
    add_test(NAME ima_quote_offline COMMAND "${Python3_EXECUTABLE}"
      "${CMAKE_SOURCE_DIR}/scripts/verificar-ima-quote.py"
      --inspector $<TARGET_FILE:lab_quote_inspect> --replayer $<TARGET_FILE:lab_ima_replay>)
    add_test(NAME session_admission COMMAND "${Python3_EXECUTABLE}"
      "${CMAKE_SOURCE_DIR}/scripts/verificar-admision.py"
      --inspector $<TARGET_FILE:lab_quote_inspect>)
    set_tests_properties(session_admission PROPERTIES TIMEOUT 45)
    set_tests_properties(ima_quote_offline PROPERTIES TIMEOUT 30)
    set_tests_properties(attestation_offline PROPERTIES
      WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" TIMEOUT 30)
  endif()
endif()

option(LAB_GRAPHICS "Build the optional raylib 5.5 Linux x86_64 game viewer" OFF)
if(LAB_GRAPHICS)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
    message(FATAL_ERROR "The pinned raylib distribution requires Linux x86_64")
  endif()
  set(LAB_RAYLIB_ROOT "${CMAKE_SOURCE_DIR}/results/toolchain/raylib-5.5" CACHE PATH "Pinned local raylib distribution")
  if(NOT EXISTS "${LAB_RAYLIB_ROOT}/lib/libraylib.so.5.5.0")
    message(FATAL_ERROR "Run python3 scripts/preparar-raylib.py before enabling LAB_GRAPHICS")
  endif()
  add_library(lab_raylib SHARED IMPORTED)
  set_target_properties(lab_raylib PROPERTIES
    IMPORTED_LOCATION "${LAB_RAYLIB_ROOT}/lib/libraylib.so.5.5.0"
    INTERFACE_SYSTEM_INCLUDE_DIRECTORIES "${LAB_RAYLIB_ROOT}/include"
    INTERFACE_INCLUDE_DIRECTORIES "${LAB_RAYLIB_ROOT}/include")
  add_executable(lab_game_viewer lab/app/game_viewer.c)
  target_link_libraries(lab_game_viewer PRIVATE lab_game lab_contract lab_raylib)
endif()
